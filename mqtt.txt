#include <WiFiNINA.h>
#include <PubSubClient.h>
#include <BH1750.h>
#include <Wire.h>

#define WIFI_SSID "YOUR_WIFI_NAME"
#define WIFI_PASSWORD "YOUR_WIFI_PASSWORD"

#define MQTT_SERVER "YOUR_CLUSTER.s1.eu.hivemq.cloud"
#define MQTT_PORT 8883
#define MQTT_TOPIC "terrarium/sunlight"

#define MQTT_USER "YOUR_HIVEMQ_USERNAME"
#define MQTT_PASSWORD "YOUR_HIVEMQ_PASSWORD"

const float lightThreshold = 50.0;

BH1750 lightSensor;
WiFiSSLClient wifiClient;
PubSubClient mqttClient(wifiClient);

bool sunlightOn = false;

void connectWiFi()
{
  Serial.print("Connecting to WiFi");

  WiFi.begin(WIFI_SSID, WIFI_PASSWORD);

  while (WiFi.status() != WL_CONNECTED)
  {
    delay(500);
    Serial.print(".");
  }

  Serial.println();
  Serial.println("WiFi connected!");

  Serial.print("IP Address: ");
  Serial.println(WiFi.localIP());
}

void connectMQTT()
{
  while (!mqttClient.connected())
  {
    Serial.print("Connecting to HiveMQ Cloud...");

    if (mqttClient.connect("ArduinoNano33IoT", MQTT_USER, MQTT_PASSWORD))
    {
      Serial.println(" connected!");
    }
    else
    {
      Serial.print(" failed, state: ");
      Serial.println(mqttClient.state());
      delay(5000);
    }
  }
}

void setup()
{
  Serial.begin(9600);
  delay(1000);

  Wire.begin();

  if (lightSensor.begin())
  {
    Serial.println("BH1750 initialized successfully!");
  }
  else
  {
    Serial.println("BH1750 initialization failed!");
    while (1);
  }

  connectWiFi();

  mqttClient.setServer(MQTT_SERVER, MQTT_PORT);

  connectMQTT();

  Serial.println("System ready!");
}

void loop()
{
  if (WiFi.status() != WL_CONNECTED)
  {
    connectWiFi();
  }

  if (!mqttClient.connected())
  {
    connectMQTT();
  }

  mqttClient.loop();

  float light = lightSensor.readLightLevel();

  Serial.print("Light level: ");
  Serial.print(light);
  Serial.println(" lux");

  if (light > lightThreshold && !sunlightOn)
  {
    if (mqttClient.publish(MQTT_TOPIC, "SUNLIGHT_START"))
    {
      Serial.println("SUNLIGHT_START sent");
    }

    sunlightOn = true;
  }
  else if (light <= lightThreshold && sunlightOn)
  {
    if (mqttClient.publish(MQTT_TOPIC, "SUNLIGHT_STOP"))
    {
      Serial.println("SUNLIGHT_STOP sent");
    }

    sunlightOn = false;
  }

  delay(1000);
}
