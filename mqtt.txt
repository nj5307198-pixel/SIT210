#include <WiFiNINA.h>
#include <PubSubClient.h>
#include <BH1750.h>
#include <Wire.h>

// Network Configuration
#define AP_SSID ""
#define AP_PASSPHRASE ""

// Broker Settings
#define BROKER_ADDRESS ""
#define BROKER_PORT 
#define TELEMETRY_TOPIC ""

#define AUTH_USER ""
#define AUTH_PASS ""

// Luminosity Trigger Value
const float LUX_THRESHOLD = 50.0;

BH1750 luxMeter;
WiFiSSLClient secureClient;
PubSubClient clientMQTT(secureClient);

bool isDaylightActive = false;   // Tracks current daylight status

void initNetworkConnection()   // Establish Wi-Fi link
{
  Serial.print("Connecting to network");

  WiFi.begin(AP_SSID, AP_PASSPHRASE);

  while (WiFi.status() != WL_CONNECTED) {
    delay(500);
    Serial.print(".");
  }

  Serial.println();
  Serial.println("Network connected successfully!");

  Serial.print("Assigned IP: ");
  Serial.println(WiFi.localIP());
}


void initBrokerLink()   // Establish MQTT session
{
  while (!clientMQTT.connected()) {

    Serial.print("Connecting to broker...");

    if (clientMQTT.connect("ArduinoNano33IoT", AUTH_USER, AUTH_PASS)) {

      Serial.println(" connected!");

    } else {

      Serial.print(" failed, error code: ");
      Serial.println(clientMQTT.state());

      delay(5000);
    }
  }
}


void setup() {

  Serial.begin(9600);

  Wire.begin();  

  if (luxMeter.begin())   // Initialize sensor
  {

    Serial.println("BH1750 sensor operational!");

  } else {

    Serial.println("BH1750 initialization error!");

    while (1);
  }

  initNetworkConnection();  // Start Wi-Fi connection

  clientMQTT.setServer(BROKER_ADDRESS, BROKER_PORT);  // Configure target broker

  initBrokerLink();  // Start MQTT session

  Serial.println("Initialization complete!");
}


void loop()
{
  if (WiFi.status() != WL_CONNECTED) initNetworkConnection();   // Verify network connectivity
  
  if (!clientMQTT.connected()) initBrokerLink();   // Verify MQTT session active

  clientMQTT.loop();

  float currentLux = luxMeter.readLightLevel();  // Acquire light intensity reading

  Serial.print("Current intensity: ");
  Serial.print(currentLux);
  Serial.println(" lux");

 // Check if light level has crossed the daylight threshold.
// isDaylightActive prevents repeated MQTT publishes while lux stays above/below the threshold.
if (currentLux > LUX_THRESHOLD && !isDaylightActive) {
  clientMQTT.publish(TELEMETRY_TOPIC, "SUNLIGHT_START");
  Serial.println("Dispatched event: SUNLIGHT_START");
  isDaylightActive = true; // latch state so we don't re-trigger every loop
}
else if (currentLux <= LUX_THRESHOLD && isDaylightActive) {
  clientMQTT.publish(TELEMETRY_TOPIC, "SUNLIGHT_STOP");
  Serial.println("Dispatched event: SUNLIGHT_STOP");
  isDaylightActive = false;
}

  delay(1000);
}
